create or replace view duit.v_transaction as
select
    cast(i.c_datetime as date) as c_date,
    i.c_datetime as c_datetime,
    i.id as id,
    i.c_username as c_username,
    i.c_detail as c_detail,
    i.c_nominal as c_nominal,
    i.c_category as c_category_id,
    c.c_name as c_category_name,
    c.c_icon as c_category_icon,
    'pemasukan' as c_status
from
    (duit.t_income i
join duit.t_category c on
    (i.c_category = c.id))
union all
select
    cast(o.c_datetime as date) as c_date,
    o.c_datetime as c_datetime,
    o.id as id,
    o.c_username as c_username,
    o.c_detail as c_detail,
    o.c_nominal as c_nominal,
    o.c_category as c_category_id,
    c.c_name as c_category_name,
    c.c_icon as c_category_icon,
    'pengeluaran' as c_status
from
    (duit.t_outcome o
join duit.t_category c on
    (o.c_category = c.id));
